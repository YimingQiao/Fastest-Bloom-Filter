cmake_minimum_required(VERSION 3.16)
project(Fastest-Bloom-Filter LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)
if(NOT CMAKE_BUILD_TYPE)
    set(CMAKE_BUILD_TYPE Release CACHE STRING "Build type" FORCE)
endif()
option(USE_AVX512 "Enable AVX-512" OFF)
option(USE_AVX2 "Enable AVX2 (required by the Impala baseline)" ON)
option(BF_NATIVE "Compile all implementations for the build machine" OFF)
option(BF_EXPERIMENTS "Build historical optimization prototypes" OFF)
option(BF_SANITIZERS "Enable address and undefined-behavior sanitizers" OFF)

find_package(Threads REQUIRED)
add_library(bf_options INTERFACE)
target_link_libraries(bf_options INTERFACE Threads::Threads)
target_include_directories(bf_options INTERFACE include)
if(BF_EXPERIMENTS)
    target_include_directories(bf_options INTERFACE experiments/2026-10-28-prototypes/include)
    target_compile_definitions(bf_options INTERFACE BF_EXPERIMENTS=1)
endif()
# Legacy sectorized implementations read uint64_t buffers through uint32_t pointers.
# Use the same aliasing setting for every implementation in the comparison.
target_compile_options(bf_options INTERFACE -fno-strict-aliasing)
if(BF_NATIVE)
    target_compile_options(bf_options INTERFACE -march=native)
elseif(USE_AVX512)
    target_compile_options(bf_options INTERFACE -mavx512f -mavx2)
elseif(USE_AVX2)
    target_compile_options(bf_options INTERFACE -mavx2)
else()
    message(FATAL_ERROR "The benchmark's Impala baseline requires AVX2")
endif()
if(BF_SANITIZERS)
    target_compile_options(bf_options INTERFACE -fsanitize=address,undefined -fno-omit-frame-pointer)
    target_link_options(bf_options INTERFACE -fsanitize=address,undefined)
endif()
foreach(name main_benchmark compare_benchmark parallel_benchmark)
    add_executable(${name} src/${name}.cpp)
    target_link_libraries(${name} PRIVATE bf_options)
endforeach()
include(CTest)
if(BUILD_TESTING)
    # The new standalone header also builds without SIMD or relaxed aliasing flags.
    add_executable(test_fast_pattern_portable tests/test_fast_pattern_portable.cpp)
    target_include_directories(test_fast_pattern_portable PRIVATE include)
    add_test(NAME portable_fast_pattern COMMAND test_fast_pattern_portable)
    add_executable(test_layouts tests/test_layouts.cpp)
    target_link_libraries(test_layouts PRIVATE bf_options)
    add_test(NAME native_layouts COMMAND test_layouts)
    add_executable(test_layouts_portable tests/test_layouts.cpp)
    target_include_directories(test_layouts_portable PRIVATE include)
    target_link_libraries(test_layouts_portable PRIVATE Threads::Threads)
    add_test(NAME portable_layouts COMMAND test_layouts_portable)
    add_executable(test_byte_split tests/test_byte_split.cpp)
    target_link_libraries(test_byte_split PRIVATE bf_options)
    add_test(NAME byte_split COMMAND test_byte_split)
    add_executable(test_byte_split_portable tests/test_byte_split.cpp)
    target_include_directories(test_byte_split_portable PRIVATE include)
    target_link_libraries(test_byte_split_portable PRIVATE Threads::Threads)
    add_test(NAME byte_split_portable COMMAND test_byte_split_portable)
    add_executable(test_parallel tests/test_parallel.cpp)
    target_link_libraries(test_parallel PRIVATE bf_options)
    add_test(NAME parallel_filters COMMAND test_parallel)
    add_executable(test_filters tests/test_filters.cpp)
    target_link_libraries(test_filters PRIVATE bf_options)
    add_test(NAME filters COMMAND test_filters)
    add_test(NAME benchmark_smoke COMMAND compare_benchmark 10 16 50 12 31 2 42)
    find_package(Python3 COMPONENTS Interpreter REQUIRED)
    add_executable(benchmark_api_test src/compare_benchmark.cpp)
    target_link_libraries(benchmark_api_test PRIVATE bf_options)
    target_include_directories(benchmark_api_test PRIVATE tests)
    target_compile_definitions(benchmark_api_test PRIVATE BF_EXTERNAL_ONLY BF_EXTERNAL_HEADER="raw_benchmark_adapter.h")
    add_test(NAME benchmark_raw_api COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/test_raw_benchmark.py $<TARGET_FILE:benchmark_api_test>)
    add_executable(benchmark_query_validation_test src/compare_benchmark.cpp)
    target_link_libraries(benchmark_query_validation_test PRIVATE bf_options)
    target_include_directories(benchmark_query_validation_test PRIVATE tests)
    target_compile_definitions(benchmark_query_validation_test PRIVATE BF_EXTERNAL_ONLY BF_EXTERNAL_HEADER="query_validation_adapter.h")
    add_test(NAME benchmark_selection COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/test_benchmark_selection.py $<TARGET_FILE:compare_benchmark> $<TARGET_FILE:benchmark_query_validation_test>)
    add_test(NAME comparison_runner COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/test_comparison_runner.py)
    add_test(NAME parallel_selection COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/test_parallel_selection.py $<TARGET_FILE:parallel_benchmark>)
    add_test(NAME external_ranking COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/test_external_ranking.py)
    add_test(NAME benchmark_accounting COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/test_benchmark.py $<TARGET_FILE:compare_benchmark>)
    add_test(NAME benchmark_reject_bad_args COMMAND compare_benchmark 99)
    set_tests_properties(benchmark_reject_bad_args PROPERTIES WILL_FAIL TRUE)
endif()
