cmake_minimum_required(VERSION 3.16)
project(Fastest-Bloom-Filter LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)
if(NOT CMAKE_BUILD_TYPE)
    set(CMAKE_BUILD_TYPE Release CACHE STRING "Build type" FORCE)
endif()
option(USE_AVX512 "Enable AVX-512" OFF)
option(USE_AVX2 "Enable AVX2 (required by the Impala baseline)" ON)
option(BF_NATIVE "Compile all implementations for the build machine" OFF)
option(BF_SANITIZERS "Enable address and undefined-behavior sanitizers" OFF)

add_library(bf_options INTERFACE)
target_include_directories(bf_options INTERFACE include)
# Legacy sectorized implementations read uint64_t buffers through uint32_t pointers.
# Use the same aliasing setting for every implementation in the comparison.
target_compile_options(bf_options INTERFACE -fno-strict-aliasing)
if(BF_NATIVE)
    target_compile_options(bf_options INTERFACE -march=native)
elseif(USE_AVX512)
    target_compile_options(bf_options INTERFACE -mavx512f -mavx2)
elseif(USE_AVX2)
    target_compile_options(bf_options INTERFACE -mavx2)
else()
    message(FATAL_ERROR "The benchmark's Impala baseline requires AVX2")
endif()
if(BF_SANITIZERS)
    target_compile_options(bf_options INTERFACE -fsanitize=address,undefined -fno-omit-frame-pointer)
    target_link_options(bf_options INTERFACE -fsanitize=address,undefined)
endif()
foreach(name main_benchmark compare_benchmark)
    add_executable(${name} src/${name}.cpp)
    target_link_libraries(${name} PRIVATE bf_options)
endforeach()
include(CTest)
if(BUILD_TESTING)
    add_executable(test_filters tests/test_filters.cpp)
    target_link_libraries(test_filters PRIVATE bf_options)
    add_test(NAME filters COMMAND test_filters)
    add_test(NAME benchmark_smoke COMMAND compare_benchmark 10 16 50 12 31 2 42)
    find_package(Python3 COMPONENTS Interpreter REQUIRED)
    add_test(NAME benchmark_accounting COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/test_benchmark.py $<TARGET_FILE:compare_benchmark>)
    add_test(NAME benchmark_reject_bad_args COMMAND compare_benchmark 99)
    set_tests_properties(benchmark_reject_bad_args PROPERTIES WILL_FAIL TRUE)
endif()
