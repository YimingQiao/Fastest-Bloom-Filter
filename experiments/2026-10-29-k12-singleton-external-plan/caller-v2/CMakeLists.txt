cmake_minimum_required(VERSION 3.18)
project(k12_singleton_public_panel LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)
if(NOT CMAKE_BUILD_TYPE)
  set(CMAKE_BUILD_TYPE Release CACHE STRING "Build type" FORCE)
endif()
set(BF_PROFILE "native" CACHE STRING "native or minimal-f")
option(BF_SANITIZERS "ASan and UBSan correctness caller" OFF)
get_filename_component(REPOSITORY "${CMAKE_CURRENT_SOURCE_DIR}/../../.." ABSOLUTE)
# These are read-only paths from the committed d70077 consumed-input descriptor.
# The preflight validates every expected source byte before configure and tests.
# No FetchContent, vendor script, dependency build, download or main-tree include.
set(FASTFILTER "${REPOSITORY}/build/all-external-confirmation/_deps/fastfilter-src/src")
set(PAGE "${REPOSITORY}/build/all-external-confirmation/_deps/pagebloom-src")
set(BOOST "${REPOSITORY}/benchmarks/boost-external/.deps/boost_1_92_0")
set(ROCKS "${REPOSITORY}/benchmarks/rocksdb-external/vendor")
foreach(header IN ITEMS "${FASTFILTER}/bloom/simd-block.h" "${PAGE}/include/pbf.h" "${BOOST}/boost/bloom.hpp" "${ROCKS}/util/bloom_impl.h")
  if(NOT EXISTS "${header}")
    message(FATAL_ERROR "Missing frozen upstream header: ${header}")
  endif()
endforeach()
add_executable(main main.cpp)
target_include_directories(main BEFORE PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}" "${FASTFILTER}" "${PAGE}/include" "${PAGE}/src" "${BOOST}" "${ROCKS}" "${ROCKS}/include")
target_compile_definitions(main PRIVATE ROCKSDB_PLATFORM_POSIX OS_LINUX)
target_compile_options(main PRIVATE -fno-strict-aliasing)
if(BF_PROFILE STREQUAL "native")
  target_compile_options(main PRIVATE -march=native)
elseif(BF_PROFILE STREQUAL "minimal-f")
  # Fixed32/64's unchanged public baseline requires AVX2. AVX512F-only describes
  # the new candidate's 512-bit intrinsic requirement, not the entire binary.
  target_compile_options(main PRIVATE -mavx2 -mavx512f -mno-avx512vl -mno-avx512dq -mno-avx512bw -mno-avx512vbmi -mno-avx512vbmi2 -mno-avx512bitalg -mno-avx512vpopcntdq)
else()
  message(FATAL_ERROR "Unknown fixed profile")
endif()
if(BF_SANITIZERS)
  target_compile_options(main PRIVATE -O1 -g -fsanitize=address,undefined -fno-omit-frame-pointer)
  target_link_options(main PRIVATE -fsanitize=address,undefined)
endif()
