cmake_minimum_required(VERSION 3.18)
project(split64_range_native LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)
find_package(Threads REQUIRED)
option(RANGE_SANITIZERS "Address and undefined behavior checks" OFF)
set(RANGE_ISA "native" CACHE STRING "native, avx2 or scalar")
add_executable(check cpp/check.cpp)
target_include_directories(check PRIVATE cpp ../../include)
target_link_libraries(check PRIVATE Threads::Threads)
target_compile_options(check PRIVATE -O3 -fno-strict-aliasing -Wall -Wextra -Werror)
if(CMAKE_CXX_COMPILER_ID MATCHES "Clang")
  target_compile_options(check PRIVATE -Wno-bitwise-instead-of-logical)
endif()
if(RANGE_ISA STREQUAL "native")
  target_compile_options(check PRIVATE -march=native)
elseif(RANGE_ISA STREQUAL "avx2")
  target_compile_options(check PRIVATE -march=x86-64-v3)
elseif(RANGE_ISA STREQUAL "scalar")
  target_compile_options(check PRIVATE -march=x86-64 -mno-avx -mno-sse4.1 -mno-ssse3)
else()
  message(FATAL_ERROR "Unknown RANGE_ISA")
endif()
if(RANGE_SANITIZERS)
  target_compile_options(check PRIVATE -fsanitize=address,undefined -fno-omit-frame-pointer)
  target_link_options(check PRIVATE -fsanitize=address,undefined)
endif()
