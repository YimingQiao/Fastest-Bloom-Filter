cmake_minimum_required(VERSION 3.18)
project(cyclic_split64 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)
option(CYCLIC_SANITIZERS "ASan and UBSan" OFF)
set(CYCLIC_ISA "native" CACHE STRING "native, avx2, scalar")
add_executable(check cpp/check.cpp)
add_executable(calibrate cpp/calibrate.cpp)
foreach(target check calibrate)
  target_include_directories(${target} PRIVATE cpp ../../include)
  target_compile_options(${target} PRIVATE -O3 -fno-strict-aliasing -Wall -Wextra -Werror)
  if(CMAKE_CXX_COMPILER_ID MATCHES "Clang")
    target_compile_options(${target} PRIVATE -Wno-bitwise-instead-of-logical)
  endif()
  if(CYCLIC_ISA STREQUAL "native")
    target_compile_options(${target} PRIVATE -march=native)
  elseif(CYCLIC_ISA STREQUAL "avx2")
    target_compile_options(${target} PRIVATE -march=x86-64-v3)
  elseif(CYCLIC_ISA STREQUAL "scalar")
    target_compile_options(${target} PRIVATE -march=x86-64 -mno-avx -mno-sse4.1 -mno-ssse3)
  else()
    message(FATAL_ERROR "Unknown CYCLIC_ISA")
  endif()
  if(CYCLIC_SANITIZERS)
    target_compile_options(${target} PRIVATE -fsanitize=address,undefined -fno-omit-frame-pointer)
    target_link_options(${target} PRIVATE -fsanitize=address,undefined)
  endif()
endforeach()
