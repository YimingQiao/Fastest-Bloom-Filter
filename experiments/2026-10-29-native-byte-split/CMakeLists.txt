cmake_minimum_required(VERSION 3.18)
project(native_byte_split_port LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_BUILD_TYPE Release)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)
include(../../benchmarks/cpp-external/dependencies.cmake)
add_executable(compare_benchmark snapshot/compare_benchmark.cpp)
target_include_directories(compare_benchmark BEFORE PRIVATE ${CMAKE_CURRENT_SOURCE_DIR} ${CMAKE_CURRENT_SOURCE_DIR}/snapshot/include)
target_link_libraries(compare_benchmark PRIVATE FastFilterExternal::options)
target_compile_definitions(compare_benchmark PRIVATE BF_EXTERNAL_ONLY BF_EXTERNAL_HEADER="adapters.hpp")
target_compile_options(compare_benchmark PRIVATE -march=native -fno-strict-aliasing)
