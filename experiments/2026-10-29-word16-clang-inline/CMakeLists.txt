cmake_minimum_required(VERSION 3.18)
project(word16_clang_inline LANGUAGES CXX)
if(POLICY CMP0135)
  cmake_policy(SET CMP0135 NEW)
endif()
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)
if(NOT CMAKE_BUILD_TYPE)
  set(CMAKE_BUILD_TYPE Release CACHE STRING "Build type" FORCE)
endif()
option(BF_INLINE_REPRESENTATIVE "Compile a representative 19-configuration suite" OFF)
option(BF_SANITIZERS "Enable address/undefined sanitizers" OFF)
include(${CMAKE_CURRENT_SOURCE_DIR}/../../benchmarks/cpp-external/dependencies.cmake)
include(${CMAKE_CURRENT_SOURCE_DIR}/../../benchmarks/boost-external/BoostHeaders.cmake)
include(${CMAKE_CURRENT_SOURCE_DIR}/../../benchmarks/rocksdb-external/dependencies.cmake)
foreach(target benchmark fault_pure fault_raw check)
  if(target STREQUAL "check")
    add_executable(${target} check.cpp)
  else()
    add_executable(${target} benchmark.cpp)
  endif()
  target_include_directories(${target} BEFORE PRIVATE ${CMAKE_CURRENT_SOURCE_DIR}
    ${CMAKE_CURRENT_SOURCE_DIR}/../2026-10-29-external-parallel-query/snapshot/include)
  target_compile_options(${target} PRIVATE -march=native -fno-strict-aliasing)
  target_link_libraries(${target} PRIVATE FastFilterExternal::options bf_boost_baseline bf_rocksdb_baseline)
  target_compile_definitions(${target} PRIVATE BF_EXTERNAL_ONLY=1 BF_EXTERNAL_HEADER="adapters.hpp" BF_NATIVE_LAYOUTS=1 BF_ROCKSDB=1)
  if(BF_INLINE_REPRESENTATIVE OR NOT target STREQUAL "benchmark")
    target_compile_definitions(${target} PRIVATE BF_INLINE_REPRESENTATIVE=1)
  endif()
  if(BF_SANITIZERS)
    target_compile_options(${target} PRIVATE -fsanitize=address,undefined -fno-omit-frame-pointer)
    target_link_options(${target} PRIVATE -fsanitize=address,undefined)
  endif()
endforeach()
target_compile_definitions(fault_pure PRIVATE BF_INLINE_FAULT_PURE=1)
target_compile_definitions(fault_raw PRIVATE BF_INLINE_FAULT_RAW=1)
enable_testing()
add_test(NAME inline_api_contract COMMAND check)

# Preserve the independent full-bitmap/halfword oracle for the include-only port.
foreach(target word16_check word16_check_scalar)
  add_executable(${target} word16_check.cpp)
  target_include_directories(${target} BEFORE PRIVATE ${CMAKE_CURRENT_SOURCE_DIR}
    ${CMAKE_CURRENT_SOURCE_DIR}/../2026-10-29-external-parallel-query/snapshot/include)
  if(target STREQUAL "word16_check_scalar")
    target_compile_options(${target} PRIVATE -mno-avx2 -mno-avx512f)
    target_compile_definitions(${target} PRIVATE BF_WORD16_FORCE_SCALAR=1)
  else()
    target_compile_options(${target} PRIVATE -march=native)
  endif()
  if(BF_SANITIZERS)
    target_compile_options(${target} PRIVATE -fsanitize=address,undefined -fno-omit-frame-pointer)
    target_link_options(${target} PRIVATE -fsanitize=address,undefined)
  endif()
  add_test(NAME ${target} COMMAND ${target})
endforeach()
