cmake_minimum_required(VERSION 3.18)
project(buffered_partitions_experiment LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)
if(NOT CMAKE_BUILD_TYPE)
  set(CMAKE_BUILD_TYPE Release CACHE STRING "Build type" FORCE)
endif()
option(BF_SANITIZERS "Address/undefined sanitizers for correctness" OFF)
option(BF_TSAN "ThreadSanitizer for correctness" OFF)
find_package(Threads REQUIRED)
foreach(target benchmark check)
  add_executable(${target} ${target}.cpp)
  target_include_directories(${target} PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}/../snapshot/include")
  target_compile_options(${target} PRIVATE -march=native)
  target_link_libraries(${target} PRIVATE Threads::Threads)
endforeach()
if(BF_SANITIZERS)
  target_compile_options(check PRIVATE -fsanitize=address,undefined -fno-omit-frame-pointer)
  target_link_options(check PRIVATE -fsanitize=address,undefined)
endif()
if(BF_TSAN)
  target_compile_options(check PRIVATE -fsanitize=thread -fno-omit-frame-pointer)
  target_link_options(check PRIVATE -fsanitize=thread)
endif()
enable_testing()
add_test(NAME partition_ownership_and_outputs COMMAND check)
