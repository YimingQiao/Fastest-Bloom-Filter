cmake_minimum_required(VERSION 3.16)
project(page_pointer_preflight LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_EXTENSIONS ON)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)
# Compile the unchanged committed worker in its own original source directory.
add_subdirectory("${CMAKE_CURRENT_SOURCE_DIR}/.." worker)
add_executable(pointer_caller_check caller_check.cpp)
foreach(PROP INCLUDE_DIRECTORIES COMPILE_DEFINITIONS COMPILE_OPTIONS LINK_OPTIONS)
  get_target_property(VALUE page_pointer_control ${PROP})
  if(VALUE)
    set_property(TARGET pointer_caller_check PROPERTY ${PROP} "${VALUE}")
  endif()
endforeach()
