cmake_minimum_required(VERSION 3.18)
project(cyclic_split64_screen_preflight LANGUAGES CXX)
if(POLICY CMP0135)
  cmake_policy(SET CMP0135 NEW)
endif()
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)
if(NOT CMAKE_BUILD_TYPE)
  set(CMAKE_BUILD_TYPE Release CACHE STRING "Build type" FORCE)
endif()
option(BF_NATIVE "Use common host ISA" ON)
option(BF_SANITIZERS "ASan and UBSan" OFF)
get_filename_component(ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../.." ABSOLUTE)
set(PARENT "${ROOT}/experiments/2026-10-29-exact-budget-range")
include(${ROOT}/benchmarks/cpp-external/dependencies.cmake)
include(${ROOT}/benchmarks/boost-external/BoostHeaders.cmake)
include(${ROOT}/benchmarks/rocksdb-external/dependencies.cmake)
find_package(Python3 COMPONENTS Interpreter REQUIRED)
foreach(target IN ITEMS benchmark plan check fixtures)
  if(target STREQUAL "benchmark" OR target STREQUAL "fixtures")
    set(source compare_benchmark.cpp)
  elseif(target STREQUAL "range_oracle")
    set(source range-layouts/check.cpp)
  else()
    set(source ${target}.cpp)
  endif()
  add_executable(${target} ${source})
  target_include_directories(${target} BEFORE PRIVATE ${CMAKE_CURRENT_SOURCE_DIR} ${PARENT} ${PARENT}/snapshot/include)
  target_link_libraries(${target} PRIVATE FastFilterExternal::options bf_boost_baseline bf_rocksdb_baseline)
  target_compile_definitions(${target} PRIVATE BF_EXTERNAL_ONLY=1 BF_EXTERNAL_HEADER="adapters.hpp")
  if(target STREQUAL "fixtures")
    target_compile_definitions(${target} PRIVATE BF_EXACT_TESTS=1)
  endif()
  target_compile_options(${target} PRIVATE -fno-strict-aliasing)
  if(BF_NATIVE)
    target_compile_options(${target} PRIVATE -march=native)
  else()
    # The pinned Fixed32/Fixed64 API being measured requires AVX2. Native
    # filters use their genuine AVX2 fallback; this is not a scalar claim.
    target_compile_options(${target} PRIVATE -mavx2 -mno-avx512f)
  endif()
  if(BF_SANITIZERS)
    target_compile_options(${target} PRIVATE -fsanitize=address,undefined -fno-omit-frame-pointer)
    target_link_options(${target} PRIVATE -fsanitize=address,undefined)
  endif()
endforeach()
enable_testing()
add_test(NAME geometry_and_output COMMAND check)
add_test(NAME cli_modes COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/test_contract.py $<TARGET_FILE:benchmark> $<TARGET_FILE:plan> $<TARGET_FILE:fixtures>)
