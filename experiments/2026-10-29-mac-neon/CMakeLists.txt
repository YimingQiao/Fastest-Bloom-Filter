cmake_minimum_required(VERSION 3.16)
project(mac_bloom_confirmation LANGUAGES CXX)
if(POLICY CMP0135)
  cmake_policy(SET CMP0135 NEW)
endif()
if(POLICY CMP0169)
  cmake_policy(SET CMP0169 OLD)
endif()
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)
if(NOT CMAKE_BUILD_TYPE)
  set(CMAKE_BUILD_TYPE Release CACHE STRING "Build type" FORCE)
endif()
include(../../benchmarks/boost-external/BoostHeaders.cmake)
include(../../benchmarks/cpp-external/dependencies.cmake)
add_executable(compare_benchmark ../../src/compare_benchmark.cpp)
target_include_directories(compare_benchmark PRIVATE ../../include "${CMAKE_CURRENT_SOURCE_DIR}")
target_link_libraries(compare_benchmark PRIVATE bf_boost_baseline FastFilterExternal::options)
target_compile_definitions(compare_benchmark PRIVATE BF_EXTERNAL_HEADER="confirmation.hpp")
target_compile_options(compare_benchmark PRIVATE -O3 -mcpu=native -fno-strict-aliasing)
