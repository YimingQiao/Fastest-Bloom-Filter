cmake_minimum_required(VERSION 3.18)
project(independent32_range LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)
set(INDEPENDENT_ISA "native" CACHE STRING "native, avx2 or scalar")
option(INDEPENDENT_SANITIZERS "ASan and UBSan" OFF)
add_executable(check check.cpp)
target_include_directories(check PRIVATE snapshot)
target_compile_options(check PRIVATE -O3 -Wall -Wextra -Werror -fno-strict-aliasing)
if(INDEPENDENT_ISA STREQUAL "native")
  target_compile_options(check PRIVATE -march=native)
elseif(INDEPENDENT_ISA STREQUAL "avx2")
  target_compile_options(check PRIVATE -march=x86-64-v3)
elseif(INDEPENDENT_ISA STREQUAL "scalar")
  target_compile_options(check PRIVATE -march=x86-64 -mno-avx -mno-sse4.1 -mno-ssse3)
else()
  message(FATAL_ERROR "unknown ISA profile")
endif()
if(INDEPENDENT_SANITIZERS)
  target_compile_options(check PRIVATE -fsanitize=address,undefined -fno-omit-frame-pointer)
  target_link_options(check PRIVATE -fsanitize=address,undefined)
endif()
