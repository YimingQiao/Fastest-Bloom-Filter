cmake_minimum_required(VERSION 3.18)
project(independent32_range_public_panel LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)
if(NOT CMAKE_BUILD_TYPE)
  set(CMAKE_BUILD_TYPE Release CACHE STRING "Build type" FORCE)
endif()
option(BF_SANITIZERS "ASan and UBSan caller correctness" OFF)
get_filename_component(REPOSITORY "${CMAKE_CURRENT_SOURCE_DIR}/../../.." ABSOLUTE)
set(FASTFILTER "${REPOSITORY}/build/all-external-confirmation/_deps/fastfilter-src/src")
set(BOOST "${REPOSITORY}/benchmarks/boost-external/.deps/boost_1_92_0")
foreach(header IN ITEMS "${FASTFILTER}/bloom/bloom.h" "${FASTFILTER}/bloom/simd-block-fixed-fpp.h" "${BOOST}/boost/bloom.hpp")
  if(NOT EXISTS "${header}")
    message(FATAL_ERROR "Missing frozen public upstream header: ${header}")
  endif()
endforeach()
add_executable(main main.cpp)
target_include_directories(main BEFORE PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}" "${FASTFILTER}" "${BOOST}")
target_compile_options(main PRIVATE -fno-strict-aliasing -march=native)
if(BF_SANITIZERS)
  target_compile_options(main PRIVATE -O1 -g -fsanitize=address,undefined -fno-omit-frame-pointer)
  target_link_options(main PRIVATE -fsanitize=address,undefined)
endif()
