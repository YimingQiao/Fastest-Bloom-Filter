cmake_minimum_required(VERSION 3.16)
project(native_range_scaling LANGUAGES CXX)
find_package(Threads REQUIRED)
add_executable(benchmark cpp/main.cpp)
target_compile_features(benchmark PRIVATE cxx_std_17)
target_include_directories(benchmark PRIVATE cpp/native)
target_compile_options(benchmark PRIVATE -O3 -march=native -fno-strict-aliasing)
target_link_libraries(benchmark PRIVATE Threads::Threads)
option(NATIVE_SCALING_ASAN "Focused harness correctness under ASan/UBSan" OFF)
if(NATIVE_SCALING_ASAN)
 target_compile_options(benchmark PRIVATE -fsanitize=address,undefined -fno-omit-frame-pointer -g)
 target_link_options(benchmark PRIVATE -fsanitize=address,undefined)
endif()
