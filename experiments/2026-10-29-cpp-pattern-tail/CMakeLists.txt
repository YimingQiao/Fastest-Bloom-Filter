cmake_minimum_required(VERSION 3.18)
project(cpp_pattern_tail LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_BUILD_TYPE Release)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)
add_executable(compare_benchmark snapshot/compare_benchmark.cpp)
target_include_directories(compare_benchmark PRIVATE ${CMAKE_CURRENT_SOURCE_DIR} ${CMAKE_CURRENT_SOURCE_DIR}/snapshot/include)
target_compile_definitions(compare_benchmark PRIVATE BF_EXTERNAL_ONLY BF_EXTERNAL_HEADER="adapters.hpp")
target_compile_options(compare_benchmark PRIVATE -march=native -fno-strict-aliasing)
add_executable(check check.cpp)
target_include_directories(check PRIVATE ${CMAKE_CURRENT_SOURCE_DIR}/snapshot/include)
target_compile_options(check PRIVATE -march=native)
add_executable(check_scalar check.cpp)
target_include_directories(check_scalar PRIVATE ${CMAKE_CURRENT_SOURCE_DIR}/snapshot/include)
include(CTest)
add_test(NAME masked_tail_native COMMAND check)
add_test(NAME masked_tail_scalar COMMAND check_scalar)
