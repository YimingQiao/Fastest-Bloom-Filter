cmake_minimum_required(VERSION 3.16)
project(four_sector_native_confirmation LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_BUILD_TYPE Release)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)
add_executable(compare_benchmark compare_benchmark_snapshot.cpp)
target_include_directories(compare_benchmark BEFORE PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}")
target_compile_definitions(compare_benchmark PRIVATE BF_EXTERNAL_ONLY=1 BF_EXTERNAL_HEADER="adapters.hpp")
target_compile_options(compare_benchmark PRIVATE -march=native -fno-strict-aliasing)
add_executable(check check.cpp)
target_include_directories(check PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}")
target_compile_options(check PRIVATE -march=native)
