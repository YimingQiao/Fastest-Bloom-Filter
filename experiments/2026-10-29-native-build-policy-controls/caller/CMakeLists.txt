cmake_minimum_required(VERSION 3.20)
project(native_build_policy_four_arm LANGUAGES CXX)
find_package(Threads REQUIRED)
option(POLICY_ASAN "Address and undefined behavior sanitizers" OFF)
add_executable(policy_confirmation cpp/main.cpp)
target_compile_features(policy_confirmation PRIVATE cxx_std_17)
target_compile_options(policy_confirmation PRIVATE -O3 -march=native -fno-strict-aliasing)
target_link_libraries(policy_confirmation PRIVATE Threads::Threads)
if(POLICY_ASAN)
  target_compile_options(policy_confirmation PRIVATE -fsanitize=address,undefined -fno-omit-frame-pointer -g)
  target_link_options(policy_confirmation PRIVATE -fsanitize=address,undefined)
endif()
