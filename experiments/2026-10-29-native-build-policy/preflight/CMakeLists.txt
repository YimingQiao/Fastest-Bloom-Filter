cmake_minimum_required(VERSION 3.20)
project(native_build_policy_correctness LANGUAGES CXX)
find_package(Threads REQUIRED)
option(POLICY_ASAN "Address and undefined behavior sanitizers" OFF)
add_executable(policy_check cpp/check.cpp)
target_compile_features(policy_check PRIVATE cxx_std_17)
target_include_directories(policy_check PRIVATE ../../2026-10-29-split64-range-native-scaling/cpp/native)
target_compile_options(policy_check PRIVATE -O3 -march=native -fno-strict-aliasing)
target_link_libraries(policy_check PRIVATE Threads::Threads)
if(POLICY_ASAN)
  target_compile_options(policy_check PRIVATE -fsanitize=address,undefined -fno-omit-frame-pointer -g)
  target_link_options(policy_check PRIVATE -fsanitize=address,undefined)
endif()
