cmake_minimum_required(VERSION 3.18)
project(byte256_experiment LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)
set(CMAKE_BUILD_TYPE Release)
include("${CMAKE_CURRENT_SOURCE_DIR}/../../benchmarks/cpp-external/dependencies.cmake")
add_executable(compare_benchmark ../../src/compare_benchmark.cpp)
target_include_directories(compare_benchmark BEFORE PRIVATE ${CMAKE_CURRENT_SOURCE_DIR})
target_link_libraries(compare_benchmark PRIVATE FastFilterExternal::options)
target_compile_definitions(compare_benchmark PRIVATE BF_EXTERNAL_ONLY=1 BF_EXTERNAL_HEADER="adapters.hpp")
target_compile_options(compare_benchmark PRIVATE -march=native -fno-strict-aliasing)
add_executable(check check.cpp)
target_compile_options(check PRIVATE -march=native)
add_executable(check_scalar check.cpp)
target_compile_options(check_scalar PRIVATE -mno-avx2 -mno-avx512f)
option(BF_SANITIZERS "Run correctness under ASan/UBSan" OFF)
if(BF_SANITIZERS)
  foreach(t check check_scalar)
    target_compile_options(${t} PRIVATE -fsanitize=address,undefined -fno-omit-frame-pointer)
    target_link_options(${t} PRIVATE -fsanitize=address,undefined)
  endforeach()
endif()
enable_testing()
add_test(NAME native COMMAND check)
add_test(NAME scalar COMMAND check_scalar)
