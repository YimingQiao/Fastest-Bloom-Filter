cmake_minimum_required(VERSION 3.18)
project(fast_pattern_range_source_preparation LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)
if(NOT CMAKE_BUILD_TYPE)
  set(CMAKE_BUILD_TYPE Release CACHE STRING "Build type" FORCE)
endif()
set(BF_BACKEND native CACHE STRING "native, avx2, scalar, or avx512f-min")
option(BF_SANITIZERS "Enable ASan+UBSan" OFF)
add_executable(check check.cpp)
target_include_directories(check PRIVATE ${CMAKE_CURRENT_SOURCE_DIR}/source-snapshot)
target_compile_options(check PRIVATE -fno-strict-aliasing)
if(BF_BACKEND STREQUAL "native")
  target_compile_options(check PRIVATE -march=native)
elseif(BF_BACKEND STREQUAL "avx2")
  target_compile_options(check PRIVATE -mavx2 -mno-avx512f)
elseif(BF_BACKEND STREQUAL "scalar")
  target_compile_options(check PRIVATE -mno-avx2 -mno-avx512f)
elseif(BF_BACKEND STREQUAL "avx512f-min")
  target_compile_options(check PRIVATE -mavx2 -mavx512f -mno-avx512vl -mno-avx512dq -mno-avx512bw)
else()
  message(FATAL_ERROR "unknown BF_BACKEND")
endif()
if(BF_SANITIZERS)
  target_compile_options(check PRIVATE -fsanitize=address,undefined -fno-omit-frame-pointer)
  target_link_options(check PRIVATE -fsanitize=address,undefined)
endif()
enable_testing()
add_test(NAME scalar_oracle_full_bitmap COMMAND check)
