cmake_minimum_required(VERSION 3.18)
project(external_parallel_query LANGUAGES CXX)
if(POLICY CMP0135)
  cmake_policy(SET CMP0135 NEW)
endif()
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)
if(NOT CMAKE_BUILD_TYPE)
  set(CMAKE_BUILD_TYPE Release CACHE STRING "Build type" FORCE)
endif()
option(BF_MT_REPRESENTATIVE "Build only representative benchmark registrations" OFF)
option(BF_SANITIZERS "Enable address/undefined sanitizers" OFF)
option(BF_TSAN "Enable ThreadSanitizer" OFF)
find_package(Threads REQUIRED)
include(${CMAKE_CURRENT_SOURCE_DIR}/../../benchmarks/cpp-external/dependencies.cmake)
include(${CMAKE_CURRENT_SOURCE_DIR}/../../benchmarks/boost-external/BoostHeaders.cmake)
include(${CMAKE_CURRENT_SOURCE_DIR}/../../benchmarks/rocksdb-external/dependencies.cmake)
foreach(target benchmark check_adapters check_executor)
  add_executable(${target} ${target}.cpp)
  target_include_directories(${target} BEFORE PRIVATE ${CMAKE_CURRENT_SOURCE_DIR} ${CMAKE_CURRENT_SOURCE_DIR}/snapshot/include)
  target_compile_options(${target} PRIVATE -march=native -fno-strict-aliasing)
  target_link_libraries(${target} PRIVATE Threads::Threads FastFilterExternal::options bf_boost_baseline bf_rocksdb_baseline)
  target_compile_definitions(${target} PRIVATE BF_EXTERNAL_ONLY=1 BF_EXTERNAL_HEADER="adapters.hpp" BF_NATIVE_LAYOUTS=1 BF_ROCKSDB=1)
  if(BF_MT_REPRESENTATIVE)
    target_compile_definitions(${target} PRIVATE BF_MT_REPRESENTATIVE=1)
  endif()
  if(BF_SANITIZERS)
    target_compile_options(${target} PRIVATE -fsanitize=address,undefined -fno-omit-frame-pointer)
    target_link_options(${target} PRIVATE -fsanitize=address,undefined)
  endif()
  if(BF_TSAN)
    target_compile_options(${target} PRIVATE -fsanitize=thread -fno-omit-frame-pointer)
    target_link_options(${target} PRIVATE -fsanitize=thread)
  endif()
endforeach()
enable_testing()
add_test(NAME executor_contract_and_faults COMMAND check_executor)
add_test(NAME immutable_adapter_contracts COMMAND check_adapters)
set_tests_properties(executor_contract_and_faults immutable_adapter_contracts PROPERTIES TIMEOUT 60)
