cmake_minimum_required(VERSION 3.18)
project(split64_bitshuffle_ktest LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_BUILD_TYPE Release)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)
add_executable(compare_benchmark ../../src/compare_benchmark.cpp)
target_include_directories(compare_benchmark PRIVATE ${CMAKE_CURRENT_SOURCE_DIR} ${CMAKE_CURRENT_SOURCE_DIR}/../../include)
target_compile_definitions(compare_benchmark PRIVATE BF_EXTERNAL_ONLY=1 BF_EXTERNAL_HEADER="adapters.hpp")
target_compile_options(compare_benchmark PRIVATE -march=native -fno-strict-aliasing)
add_executable(check check.cpp)
target_compile_options(check PRIVATE -march=native)
add_executable(check_scalar check.cpp)
target_compile_options(check_scalar PRIVATE -mno-avx2 -mno-avx512f)
add_executable(check_avx2 check.cpp)
target_compile_options(check_avx2 PRIVATE -march=x86-64 -mavx2 -mno-avx512f)
add_executable(check_bitalg_min check.cpp)
target_compile_options(check_bitalg_min PRIVATE -march=x86-64 -mavx2 -mavx512f -mavx512bw -mavx512bitalg -mno-avx512vl -mno-avx512dq)
target_compile_definitions(check_bitalg_min PRIVATE BF_EXPECT_BITALG=1)
option(BF_SANITIZERS "Check with address/undefined sanitizers" OFF)
if(BF_SANITIZERS)
  foreach(target check check_scalar check_avx2 check_bitalg_min)
    target_compile_options(${target} PRIVATE -fsanitize=address,undefined -fno-omit-frame-pointer)
    target_link_options(${target} PRIVATE -fsanitize=address,undefined)
  endforeach()
endif()
enable_testing()
add_test(NAME native COMMAND check)
add_test(NAME scalar COMMAND check_scalar)

add_test(NAME avx2 COMMAND check_avx2)
add_test(NAME bitalg_min COMMAND check_bitalg_min)
