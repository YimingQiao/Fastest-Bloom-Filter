cmake_minimum_required(VERSION 3.18)
project(native_k12_parallel_callers LANGUAGES CXX)
if(POLICY CMP0135)
  cmake_policy(SET CMP0135 NEW)
endif()
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)
if(NOT CMAKE_BUILD_TYPE)
  set(CMAKE_BUILD_TYPE Release CACHE STRING "Build type" FORCE)
endif()
option(BF_NATIVE "Common host ISA" ON)
option(BF_SANITIZERS "Address and undefined behavior checks" OFF)
get_filename_component(ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../.." ABSOLUTE)
include(${ROOT}/benchmarks/cpp-external/dependencies.cmake)
include(${ROOT}/benchmarks/boost-external/BoostHeaders.cmake)
include(${ROOT}/benchmarks/rocksdb-external/dependencies.cmake)
find_package(Threads REQUIRED)
add_executable(query query.cpp)
add_executable(public_api public_api.cpp)
add_executable(check_executor check_executor.cpp)
add_executable(check_phase check_phase.cpp)
foreach(target query public_api check_executor check_phase)
  target_include_directories(${target} BEFORE PRIVATE ${CMAKE_CURRENT_SOURCE_DIR} ${CMAKE_CURRENT_SOURCE_DIR}/native/include)
  target_compile_options(${target} PRIVATE -fno-strict-aliasing)
  target_link_libraries(${target} PRIVATE Threads::Threads)
  if(BF_NATIVE)
    target_compile_options(${target} PRIVATE -march=native)
  else()
    target_compile_options(${target} PRIVATE -mavx2 -mno-avx512f)
  endif()
  if(BF_SANITIZERS)
    target_compile_options(${target} PRIVATE -fsanitize=address,undefined -fno-omit-frame-pointer)
    target_link_options(${target} PRIVATE -fsanitize=address,undefined)
  endif()
endforeach()
target_link_libraries(query PRIVATE FastFilterExternal::options bf_boost_baseline bf_rocksdb_baseline)
enable_testing()
add_test(NAME frozen_executor_faults COMMAND check_executor)
add_test(NAME balanced_phase_contract COMMAND check_phase)
