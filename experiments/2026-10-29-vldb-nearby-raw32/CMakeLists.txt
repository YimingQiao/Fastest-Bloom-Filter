cmake_minimum_required(VERSION 3.16)
project(vldb_nearby_raw32_public_caller LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_EXTENSIONS ON)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)
set(VLDB_BACKEND "native" CACHE STRING "native or native-asan")
get_filename_component(REPO "${CMAKE_CURRENT_SOURCE_DIR}/../.." ABSOLUTE)
set(PARENT "${REPO}/benchmarks/vldb-original-external")
set(SHARED "${REPO}/benchmarks/vldb-raw32-stage3")
set(DTL "${PARENT}/.deps/dtl")
set(LEGACY_BOOST_ROOT "$ENV{HOME}/.local/boost-1.83/include" CACHE PATH "One whole Boost1.83 root")
if(NOT CMAKE_CXX_COMPILER_ID STREQUAL "GNU" OR NOT CMAKE_CXX_COMPILER_VERSION VERSION_EQUAL "12.2.0")
  message(FATAL_ERROR "Reuse original objects only with verified GCC12.2")
endif()
if(VLDB_BACKEND STREQUAL "native")
  set(FLAGS -O3 -march=native -fno-strict-aliasing)
elseif(VLDB_BACKEND STREQUAL "native-asan")
  set(FLAGS -O1 -march=native -fno-strict-aliasing -fsanitize=address,undefined -fno-sanitize-recover=all -fno-omit-frame-pointer)
else()
  message(FATAL_ERROR "Only tested original library profiles are available")
endif()
set(VLDB_ARCHIVE "" CACHE FILEPATH "Explicit previously verified original archive")
set(VLDB_ARCHIVE_SHA256 "" CACHE STRING "Required original archive hash")
if(NOT EXISTS "${VLDB_ARCHIVE}")
  message(FATAL_ERROR "Original archive missing; no automatic upstream rebuild")
endif()
file(SHA256 "${VLDB_ARCHIVE}" ACTUAL_ARCHIVE_SHA256)
if(NOT VLDB_ARCHIVE_SHA256 STREQUAL ACTUAL_ARCHIVE_SHA256)
  message(FATAL_ERROR "Original archive identity missing or changed")
endif()
find_package(Threads REQUIRED)
add_library(vldb_reused STATIC IMPORTED)
set_target_properties(vldb_reused PROPERTIES IMPORTED_LOCATION "${VLDB_ARCHIVE}")
add_executable(nearby_raw32 worker.cpp)
add_executable(nearby_check check.cpp)
foreach(TARGET nearby_raw32 nearby_check)
  target_include_directories(${TARGET} BEFORE PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}" "${SHARED}" "${SHARED}/native-snapshot/include" "${PARENT}" "${DTL}/src" "${DTL}/thirdparty" "${LEGACY_BOOST_ROOT}")
  target_compile_options(${TARGET} PRIVATE ${FLAGS} -Wno-ignored-attributes -include "${PARENT}/compat.hpp")
  target_link_libraries(${TARGET} PRIVATE vldb_reused Threads::Threads ${CMAKE_DL_LIBS})
  if(VLDB_BACKEND STREQUAL "native-asan")
    target_link_options(${TARGET} PRIVATE -fsanitize=address,undefined)
  endif()
endforeach()
