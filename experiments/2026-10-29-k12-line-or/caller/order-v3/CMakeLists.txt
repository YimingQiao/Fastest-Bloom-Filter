cmake_minimum_required(VERSION 3.18)
project(k12_line_or_order LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)
if(NOT CMAKE_BUILD_TYPE)
  set(CMAKE_BUILD_TYPE Release CACHE STRING "Build type" FORCE)
endif()
option(BF_SANITIZERS "ASan and UBSan" OFF)
add_executable(compare_benchmark compare_benchmark.cpp)
target_include_directories(compare_benchmark PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}/..")
target_compile_options(compare_benchmark PRIVATE -fno-strict-aliasing -march=native)
if(BF_SANITIZERS)
  target_compile_options(compare_benchmark PRIVATE -fsanitize=address,undefined -fno-sanitize-recover=all -fno-omit-frame-pointer)
  target_link_options(compare_benchmark PRIVATE -fsanitize=address,undefined)
endif()
