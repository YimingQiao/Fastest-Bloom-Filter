cmake_minimum_required(VERSION 3.18)
project(k12_line_or LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)
if(NOT CMAKE_BUILD_TYPE)
  set(CMAKE_BUILD_TYPE Release CACHE STRING "Build type" FORCE)
endif()
set(BF_ISA "native" CACHE STRING "native or minimal-f")
option(BF_SANITIZERS "ASan and UBSan" OFF)
if(NOT BF_ISA STREQUAL "native" AND NOT BF_ISA STREQUAL "minimal-f")
  message(FATAL_ERROR "Unknown ISA profile")
endif()
add_executable(compare_benchmark compare_benchmark.cpp)
add_executable(check check.cpp)
foreach(target compare_benchmark check)
  target_compile_options(${target} PRIVATE -fno-strict-aliasing)
  if(BF_ISA STREQUAL "native")
    target_compile_options(${target} PRIVATE -march=native)
  else()
    target_compile_options(${target} PRIVATE -march=x86-64 -mavx512f -mno-avx512vl -mno-avx512dq -mno-avx512bw -mno-avx512bitalg -mno-avx512vbmi)
  endif()
  if(BF_SANITIZERS)
    target_compile_options(${target} PRIVATE -fsanitize=address,undefined -fno-sanitize-recover=all -fno-omit-frame-pointer)
    target_link_options(${target} PRIVATE -fsanitize=address,undefined)
  endif()
endforeach()
enable_testing()
add_test(NAME complete_bitmap_and_guards COMMAND check)
