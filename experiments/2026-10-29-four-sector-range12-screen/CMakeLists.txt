cmake_minimum_required(VERSION 3.18)
project(four_sector_range12_screen LANGUAGES CXX)
if(POLICY CMP0135)
  cmake_policy(SET CMP0135 NEW)
endif()
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)
if(NOT CMAKE_BUILD_TYPE)
  set(CMAKE_BUILD_TYPE Release CACHE STRING "Build type" FORCE)
endif()
option(BF_NATIVE "Use native ISA" ON)
option(BF_SANITIZERS "Use ASan and UBSan" OFF)
get_filename_component(ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../.." ABSOLUTE)
set(PARENT "${ROOT}/experiments/2026-10-29-split64-range-parallel-screen")
include(${ROOT}/benchmarks/cpp-external/dependencies.cmake)
include(${ROOT}/benchmarks/boost-external/BoostHeaders.cmake)
include(${ROOT}/benchmarks/rocksdb-external/dependencies.cmake)
add_executable(benchmark compare_benchmark.cpp)
target_include_directories(benchmark BEFORE PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}" "${PARENT}" "${PARENT}/snapshot/include")
target_link_libraries(benchmark PRIVATE FastFilterExternal::options bf_boost_baseline bf_rocksdb_baseline)
target_compile_definitions(benchmark PRIVATE BF_EXTERNAL_ONLY=1 BF_EXTERNAL_HEADER="adapters.hpp")
target_compile_options(benchmark PRIVATE -fno-strict-aliasing)
if(BF_NATIVE)
  target_compile_options(benchmark PRIVATE -march=native)
else()
  target_compile_options(benchmark PRIVATE -mavx2 -mno-avx512f)
endif()
if(BF_SANITIZERS)
  target_compile_options(benchmark PRIVATE -fsanitize=address,undefined -fno-sanitize-recover=all -fno-omit-frame-pointer)
  target_link_options(benchmark PRIVATE -fsanitize=address,undefined)
endif()
