cmake_minimum_required(VERSION 3.18)
project(exact_range_shared_query LANGUAGES CXX)
if(POLICY CMP0135)
  cmake_policy(SET CMP0135 NEW)
endif()
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)
if(NOT CMAKE_BUILD_TYPE)
  set(CMAKE_BUILD_TYPE Release CACHE STRING "Build type" FORCE)
endif()
option(BF_NATIVE "Use common host ISA" ON)
option(BF_SANITIZERS "Enable ASan/UBSan correctness instrumentation" OFF)
option(BF_TSAN "Enable ThreadSanitizer correctness instrumentation" OFF)
if(BF_SANITIZERS AND BF_TSAN)
  message(FATAL_ERROR "Use separate ASan and TSan build directories")
endif()
get_filename_component(ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../.." ABSOLUTE)
include(${ROOT}/benchmarks/cpp-external/dependencies.cmake)
include(${ROOT}/benchmarks/boost-external/BoostHeaders.cmake)
include(${ROOT}/benchmarks/rocksdb-external/dependencies.cmake)
find_package(Threads REQUIRED)
foreach(target benchmark plan check_geometry check_adapters check_executor)
  add_executable(${target} ${target}.cpp)
  target_include_directories(${target} BEFORE PRIVATE
    ${CMAKE_CURRENT_SOURCE_DIR} ${CMAKE_CURRENT_SOURCE_DIR}/snapshot/include)
  target_compile_options(${target} PRIVATE -fno-strict-aliasing)
  target_compile_definitions(${target} PRIVATE BF_EXTERNAL_ONLY=1 BF_EXTERNAL_HEADER="adapters.hpp")
  target_link_libraries(${target} PRIVATE Threads::Threads
    FastFilterExternal::options bf_boost_baseline bf_rocksdb_baseline)
  if(BF_NATIVE)
    target_compile_options(${target} PRIVATE -march=native)
  else()
    # Pinned FastFilter Fixed32/64 public APIs require AVX2. This is not
    # a scalar external-baseline claim; native scalar tests remain separate.
    target_compile_options(${target} PRIVATE -mavx2 -mno-avx512f)
  endif()
  if(BF_SANITIZERS)
    target_compile_options(${target} PRIVATE -fsanitize=address,undefined -fno-omit-frame-pointer)
    target_link_options(${target} PRIVATE -fsanitize=address,undefined)
  endif()
  if(BF_TSAN)
    target_compile_options(${target} PRIVATE -fsanitize=thread -fno-omit-frame-pointer)
    target_link_options(${target} PRIVATE -fsanitize=thread)
  endif()
endforeach()
enable_testing()
add_test(NAME geometry_and_complete_outputs COMMAND check_geometry)
add_test(NAME concurrent_complete_outputs COMMAND check_adapters)
add_test(NAME executor_contract_and_faults COMMAND check_executor)
set_tests_properties(geometry_and_complete_outputs concurrent_complete_outputs
  executor_contract_and_faults PROPERTIES TIMEOUT 180)
