cmake_minimum_required(VERSION 3.18)
project(range_parallel_additive_preflight LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)
if(NOT CMAKE_BUILD_TYPE)
  set(CMAKE_BUILD_TYPE Release CACHE STRING "Build type" FORCE)
endif()
find_package(Threads REQUIRED)
# Build the immutable initial sources in their own subdirectory.
add_subdirectory(.. original)
foreach(target allocation_actual dense_current dense_reference native_parity)
  if(target STREQUAL "allocation_actual")
    set(source allocation.cpp)
  elseif(target STREQUAL "native_parity")
    set(source native_parity.cpp)
  else()
    set(source dense.cpp)
  endif()
  add_executable(${target} ${source})
  foreach(property COMPILE_OPTIONS COMPILE_DEFINITIONS INCLUDE_DIRECTORIES LINK_LIBRARIES)
    get_target_property(values benchmark ${property})
    if(values)
      set_property(TARGET ${target} PROPERTY ${property} "${values}")
    endif()
  endforeach()
  get_target_property(link_options benchmark LINK_OPTIONS)
  if(link_options)
    target_link_options(${target} PRIVATE ${link_options})
  endif()
  if(target STREQUAL "dense_reference")
    target_compile_definitions(${target} PRIVATE BF_DENSE_REFERENCE=1)
    target_include_directories(${target} BEFORE PRIVATE
      ${CMAKE_CURRENT_SOURCE_DIR}/../../2026-10-29-exact-budget-range
      ${CMAKE_CURRENT_SOURCE_DIR}/../../2026-10-29-exact-budget-range/snapshot/include)
  endif()
endforeach()
