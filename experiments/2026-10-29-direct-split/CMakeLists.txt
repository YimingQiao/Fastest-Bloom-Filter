cmake_minimum_required(VERSION 3.18)
project(direct_split_experiment LANGUAGES CXX)
if(POLICY CMP0135)
  cmake_policy(SET CMP0135 NEW)
endif()
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)
if(NOT CMAKE_BUILD_TYPE)
  set(CMAKE_BUILD_TYPE Release CACHE STRING "Build type" FORCE)
endif()
option(BF_SANITIZERS "Enable ASan and UBSan for correctness" OFF)
include(../../benchmarks/cpp-external/dependencies.cmake)
add_executable(compare_benchmark ../../src/compare_benchmark.cpp)
target_include_directories(compare_benchmark BEFORE PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}" ../../include)
target_link_libraries(compare_benchmark PRIVATE FastFilterExternal::options)
target_compile_definitions(compare_benchmark PRIVATE BF_EXTERNAL_ONLY=1 BF_EXTERNAL_HEADER="adapters.hpp")
target_compile_options(compare_benchmark PRIVATE -march=native -fno-strict-aliasing)
add_executable(check check.cpp)
target_include_directories(check PRIVATE ../../include)
target_compile_options(check PRIVATE -march=native)
add_executable(check_scalar check.cpp)
target_include_directories(check_scalar PRIVATE ../../include)
target_compile_options(check_scalar PRIVATE -mno-avx2 -mno-avx512f -mno-sse4.1 -mno-ssse3)
if(BF_SANITIZERS)
  foreach(target check check_scalar)
    target_compile_options(${target} PRIVATE -fsanitize=address,undefined -fno-omit-frame-pointer)
    target_link_options(${target} PRIVATE -fsanitize=address,undefined)
  endforeach()
endif()
enable_testing()
add_test(NAME direct_split_native_reference COMMAND check)
add_test(NAME direct_split_scalar_reference COMMAND check_scalar)
