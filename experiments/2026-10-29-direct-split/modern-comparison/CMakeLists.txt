cmake_minimum_required(VERSION 3.18)
project(direct_split_modern_comparison LANGUAGES CXX)
if(POLICY CMP0135)
  cmake_policy(SET CMP0135 NEW)
endif()
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)
if(NOT CMAKE_BUILD_TYPE)
  set(CMAKE_BUILD_TYPE Release CACHE STRING "Build type" FORCE)
endif()
include(../../../benchmarks/cpp-external/dependencies.cmake)
include(../../../benchmarks/rocksdb-external/dependencies.cmake)
add_executable(compare_benchmark snapshot/compare_benchmark.cpp)
target_include_directories(compare_benchmark BEFORE PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}" "${CMAKE_CURRENT_SOURCE_DIR}/snapshot/include")
target_link_libraries(compare_benchmark PRIVATE FastFilterExternal::options bf_rocksdb_baseline)
target_compile_definitions(compare_benchmark PRIVATE BF_EXTERNAL_ONLY=1 BF_EXTERNAL_HEADER="adapters.hpp")
target_compile_options(compare_benchmark PRIVATE -march=native -fno-strict-aliasing -Werror=macro-redefined)
