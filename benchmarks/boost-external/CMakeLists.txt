cmake_minimum_required(VERSION 3.16)
project(fastest_bloom_boost_baseline LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)
if(NOT CMAKE_BUILD_TYPE)
  set(CMAKE_BUILD_TYPE Release CACHE STRING "Build type" FORCE)
endif()
option(BF_NATIVE "Compile all filters for the current host" ON)
option(BF_SANITIZERS "Address and undefined-behavior sanitizers" OFF)
include(BoostHeaders.cmake)
add_executable(compare_benchmark ../../src/compare_benchmark.cpp)
target_link_libraries(compare_benchmark PRIVATE bf_boost_baseline)
target_include_directories(compare_benchmark PRIVATE ../../include)
target_compile_definitions(compare_benchmark PRIVATE BF_EXTERNAL_ONLY=1 BF_EXTERNAL_HEADER="adapters.hpp")
target_compile_options(compare_benchmark PRIVATE -fno-strict-aliasing)
string(TOLOWER "${CMAKE_SYSTEM_PROCESSOR}" BF_PROCESSOR)
if(BF_NATIVE AND BF_PROCESSOR MATCHES "^(arm64|aarch64)$")
  target_compile_options(compare_benchmark PRIVATE -mcpu=native)
elseif(BF_NATIVE)
  target_compile_options(compare_benchmark PRIVATE -march=native)
elseif(BF_PROCESSOR MATCHES "^(x86_64|amd64|i[3-6]86)$")
  target_compile_options(compare_benchmark PRIVATE -mavx2)
endif()
if(BF_SANITIZERS)
  target_compile_options(compare_benchmark PRIVATE -fsanitize=address,undefined -fno-omit-frame-pointer)
  target_link_options(compare_benchmark PRIVATE -fsanitize=address,undefined)
endif()
