cmake_minimum_required(VERSION 3.16)
project(vldb_raw32_paired_workers LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_EXTENSIONS ON)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)
set(VLDB_BACKEND "native" CACHE STRING "native or native-asan")
get_filename_component(REPO "${CMAKE_CURRENT_SOURCE_DIR}/../../.." ABSOLUTE)
set(PARENT "${REPO}/benchmarks/vldb-original-external")
set(BASE "${REPO}/benchmarks/vldb-raw32-stage3")
set(DTL "${PARENT}/.deps/dtl")
set(LEGACY_BOOST_ROOT "$ENV{HOME}/.local/boost-1.83/include" CACHE PATH "Complete original Boost1.83 root")
set(MODERN_BOOST_ROOT "${REPO}/benchmarks/boost-external/.deps/boost_1_92_0" CACHE PATH "Complete modern Boost1.92 root")
set(FASTFILTER_ROOT "${REPO}/build/all-external-confirmation/_deps/fastfilter-src" CACHE PATH "Existing pinned FastFilter headers")
set(PAGE_ROOT "${REPO}/build/all-external-confirmation/_deps/pagebloom-src" CACHE PATH "Existing pinned PageBloom sources")
set(ROCKS_ROOT "${REPO}/benchmarks/rocksdb-external/vendor")
if(NOT CMAKE_CXX_COMPILER_ID STREQUAL "GNU" OR NOT CMAKE_CXX_COMPILER_VERSION VERSION_EQUAL "12.2.0")
  message(FATAL_ERROR "Use the verified GCC12.2 toolchain for both workers")
endif()
if(VLDB_BACKEND STREQUAL "native")
  set(DEFAULT_ARCHIVE "${REPO}/build/vldb-public-preflight-v2-gcc/native/libvldb_pinned.a")
  set(FLAGS -O3 -march=native -fno-strict-aliasing)
elseif(VLDB_BACKEND STREQUAL "native-asan")
  set(DEFAULT_ARCHIVE "${REPO}/build/vldb-public-preflight-v5-gcc/native-asan/libvldb_pinned.a")
  set(FLAGS -O1 -march=native -fno-strict-aliasing -fsanitize=address,undefined -fno-sanitize-recover=all -fno-omit-frame-pointer)
else()
  message(FATAL_ERROR "Only the native worker and its matching instrumented proof are planned")
endif()
set(VLDB_ARCHIVE "${DEFAULT_ARCHIVE}" CACHE FILEPATH "Exact previously verified upstream archive")
set(VLDB_ARCHIVE_SHA256 "" CACHE STRING "Required reviewed archive hash")
if(NOT EXISTS "${VLDB_ARCHIVE}")
  message(FATAL_ERROR "Pinned archive absent; original sources are never rebuilt here")
endif()
file(SHA256 "${VLDB_ARCHIVE}" ACTUAL_ARCHIVE_SHA256)
if(NOT VLDB_ARCHIVE_SHA256 STREQUAL ACTUAL_ARCHIVE_SHA256)
  message(FATAL_ERROR "Pinned archive digest missing or wrong")
endif()
find_package(Threads REQUIRED)
add_library(original_vldb STATIC IMPORTED)
set_target_properties(original_vldb PROPERTIES IMPORTED_LOCATION "${VLDB_ARCHIVE}")
set(WORKERS modern)
if(VLDB_BACKEND STREQUAL "native-asan")
  list(APPEND WORKERS legacy)
endif()
foreach(TARGET ${WORKERS})
  if("${TARGET}" STREQUAL "modern")
    add_executable(${TARGET} modern.cpp)
  else()
    add_executable(${TARGET} "${BASE}/legacy.cpp")
  endif()
  target_include_directories(${TARGET} BEFORE PRIVATE "${BASE}" "${BASE}/native-snapshot/include")
  target_compile_options(${TARGET} PRIVATE ${FLAGS})
  if(VLDB_BACKEND STREQUAL "native-asan")
    target_link_options(${TARGET} PRIVATE -fsanitize=address,undefined)
  endif()
endforeach()
if(TARGET legacy)
target_include_directories(legacy PRIVATE "${PARENT}" "${DTL}/src" "${DTL}/thirdparty" "${LEGACY_BOOST_ROOT}")
target_compile_options(legacy PRIVATE -include "${PARENT}/compat.hpp" -Wno-ignored-attributes)
target_link_libraries(legacy PRIVATE original_vldb Threads::Threads ${CMAKE_DL_LIBS})
endif()
target_include_directories(modern PRIVATE "${MODERN_BOOST_ROOT}" "${FASTFILTER_ROOT}/src" "${PAGE_ROOT}/include" "${PAGE_ROOT}/src" "${ROCKS_ROOT}" "${ROCKS_ROOT}/include")
target_compile_definitions(modern PRIVATE ROCKSDB_PLATFORM_POSIX OS_LINUX)
if(VLDB_BACKEND STREQUAL "native")
  add_executable(large_accounting "${BASE}/accounting.cpp" "${PARENT}/allocation_audit.cpp")
  target_include_directories(large_accounting BEFORE PRIVATE "${BASE}" "${BASE}/native-snapshot/include")
  target_include_directories(large_accounting PRIVATE "${PARENT}" "${DTL}/src" "${DTL}/thirdparty" "${LEGACY_BOOST_ROOT}")
  target_compile_options(large_accounting PRIVATE ${FLAGS} -include "${PARENT}/compat.hpp" -Wno-ignored-attributes)
  target_compile_definitions(large_accounting PRIVATE VLDB_ALLOCATION_AUDIT=1)
  target_link_options(large_accounting PRIVATE -Wl,--wrap=mmap -Wl,--wrap=munmap)
  target_link_libraries(large_accounting PRIVATE original_vldb Threads::Threads ${CMAKE_DL_LIBS})
endif()
