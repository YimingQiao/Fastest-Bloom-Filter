cmake_minimum_required(VERSION 3.16)
project(vldb_original_public_api LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_EXTENSIONS ON)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)
set(VLDB_BACKEND "native" CACHE STRING "native, avx2, scalar")
option(VLDB_SANITIZERS "Uninstrumented public API with ASan/UBSan" OFF)
option(VLDB_ACCOUNTING "Separate allocation-accounting correctness target" ON)
get_filename_component(REPO "${CMAKE_CURRENT_SOURCE_DIR}/../.." ABSOLUTE)
set(DTL "${CMAKE_CURRENT_SOURCE_DIR}/.deps/dtl")
if(NOT EXISTS "${DTL}/src/dtl/filter/bbf_32.cpp")
  message(FATAL_ERROR "Run fetch.py first; dependencies must match source-lock.json")
endif()
find_package(Boost REQUIRED)
find_package(Threads REQUIRED)
function(vldb_options TARGET)
  target_compile_options(${TARGET} PRIVATE -O3)
  if(VLDB_BACKEND STREQUAL "native")
    target_compile_options(${TARGET} PRIVATE -march=native)
  elseif(VLDB_BACKEND STREQUAL "avx2")
    target_compile_options(${TARGET} PRIVATE -march=x86-64-v3)
  elseif(VLDB_BACKEND STREQUAL "scalar")
    target_compile_options(${TARGET} PRIVATE -march=x86-64 -mno-avx -mno-sse4.1 -mno-ssse3)
  else()
    message(FATAL_ERROR "Unknown VLDB_BACKEND")
  endif()
  if(VLDB_SANITIZERS)
    # Correctness-only instrumentation: keep template-heavy historical code
    # debuggable and bound compiler memory; ordinary targets remain exact -O3.
    target_compile_options(${TARGET} PRIVATE -O1 -fsanitize=address,undefined -fno-sanitize-recover=all -fno-omit-frame-pointer)
    target_link_options(${TARGET} PRIVATE -fsanitize=address,undefined)
  endif()
endfunction()
set(SOURCES "${DTL}/src/dtl/filter/bbf_32.cpp" "${DTL}/src/dtl/filter/bbf_64.cpp"
            "${DTL}/src/dtl/filter/model/calibration_data.cpp" "${DTL}/src/dtl/filter/platform.cpp")
foreach(WORD 32 64)
  foreach(W 1 2 4 8 16)
    list(APPEND SOURCES "${DTL}/src/dtl/filter/blocked_bloomfilter/instances/blocked_bloomfilter_logic_u${WORD}_w${W}.cpp")
  endforeach()
endforeach()
add_library(vldb_pinned STATIC ${SOURCES})
target_include_directories(vldb_pinned PUBLIC "${DTL}/src" "${DTL}/thirdparty" ${Boost_INCLUDE_DIRS})
target_compile_options(vldb_pinned PRIVATE -include "${CMAKE_CURRENT_SOURCE_DIR}/compat.hpp" -Wno-ignored-attributes)
target_link_libraries(vldb_pinned PUBLIC Threads::Threads ${CMAKE_DL_LIBS})
vldb_options(vldb_pinned)
# No FAST_BUILD macro, global calibration benchmark, NUMA policy, or upstream edits.
add_executable(vldb_public_check check.cpp)
target_include_directories(vldb_public_check PRIVATE "${REPO}/include")
target_link_libraries(vldb_public_check PRIVATE vldb_pinned)
vldb_options(vldb_public_check)
enable_testing()
foreach(CASE sector32-u1 sector64-u1 register64-u0 sector32-u2 sector32-u4 sector64-u2 sector64-u4 native)
  add_test(NAME vldb_public_${CASE} COMMAND vldb_public_check ${CASE})
endforeach()
# The prefix tracker changes redzone placement. Keep it out of the ordinary
# target, and never substitute it for the unmodified allocator ASan proof.
if(VLDB_ACCOUNTING AND NOT VLDB_SANITIZERS)
  add_executable(vldb_accounting_check accounting_check.cpp allocation_audit.cpp)
  target_include_directories(vldb_accounting_check PRIVATE "${REPO}/include")
  target_compile_definitions(vldb_accounting_check PRIVATE VLDB_ALLOCATION_AUDIT=1)
  target_link_options(vldb_accounting_check PRIVATE -Wl,--wrap=mmap -Wl,--wrap=munmap)
  target_link_libraries(vldb_accounting_check PRIVATE vldb_pinned)
  vldb_options(vldb_accounting_check)
  foreach(CASE sector32-u1 sector64-u1 register64-u0)
    add_test(NAME vldb_accounting_${CASE} COMMAND vldb_accounting_check ${CASE})
  endforeach()
endif()
