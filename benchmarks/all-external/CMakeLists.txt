cmake_minimum_required(VERSION 3.18)
project(fastest_bloom_all_external LANGUAGES CXX)
if(POLICY CMP0135)
  cmake_policy(SET CMP0135 NEW)
endif()
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)
if(NOT CMAKE_BUILD_TYPE)
  set(CMAKE_BUILD_TYPE Release CACHE STRING "Build type" FORCE)
endif()
option(BF_NATIVE "Give every candidate the same host instruction set" ON)
option(BF_SANITIZERS "Enable ASan and UBSan" OFF)
option(BF_WIDE_EXPERIMENTS "Include experimental AVX-512 64/128-entry dictionaries" OFF)
option(BF_LAYOUT_EXPERIMENTS "Include experimental four-sector and 64-bit split-block layouts" OFF)
include(${CMAKE_CURRENT_SOURCE_DIR}/../cpp-external/dependencies.cmake)
include(${CMAKE_CURRENT_SOURCE_DIR}/../boost-external/BoostHeaders.cmake)
add_executable(compare_benchmark ../../src/compare_benchmark.cpp)
target_include_directories(compare_benchmark BEFORE PRIVATE ${CMAKE_CURRENT_SOURCE_DIR})
target_link_libraries(compare_benchmark PRIVATE FastFilterExternal::options bf_boost_baseline)
target_compile_definitions(compare_benchmark PRIVATE BF_EXTERNAL_ONLY=1 BF_EXTERNAL_HEADER="adapters.hpp")
if(BF_WIDE_EXPERIMENTS)
  target_compile_definitions(compare_benchmark PRIVATE BF_WIDE_EXPERIMENTS=1)
endif()
if(BF_LAYOUT_EXPERIMENTS)
  target_compile_definitions(compare_benchmark PRIVATE BF_LAYOUT_EXPERIMENTS=1)
endif()
target_compile_options(compare_benchmark PRIVATE -fno-strict-aliasing)
if(BF_NATIVE)
  target_compile_options(compare_benchmark PRIVATE -march=native)
else()
  target_compile_options(compare_benchmark PRIVATE -mavx2)
endif()
if(BF_SANITIZERS)
  target_compile_options(compare_benchmark PRIVATE -fsanitize=address,undefined -fno-omit-frame-pointer)
  target_link_options(compare_benchmark PRIVATE -fsanitize=address,undefined)
endif()
configure_file(upstreams.json ${CMAKE_CURRENT_BINARY_DIR}/upstreams.json COPYONLY)
