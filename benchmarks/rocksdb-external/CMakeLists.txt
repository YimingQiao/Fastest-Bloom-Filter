cmake_minimum_required(VERSION 3.18)
project(fastest_bloom_rocksdb_external LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)
if(NOT CMAKE_BUILD_TYPE)
  set(CMAKE_BUILD_TYPE Release CACHE STRING "Build type" FORCE)
endif()
option(BF_NATIVE "Use the host instruction set for every candidate" ON)
option(BF_SANITIZERS "Enable AddressSanitizer and UndefinedBehaviorSanitizer" OFF)
include(dependencies.cmake)
add_executable(compare_benchmark ../../src/compare_benchmark.cpp)
add_executable(check check.cpp)
foreach(target compare_benchmark check)
  target_include_directories(${target} BEFORE PRIVATE ${CMAKE_CURRENT_SOURCE_DIR})
  target_link_libraries(${target} PRIVATE bf_rocksdb_baseline)
  target_compile_options(${target} PRIVATE -fno-strict-aliasing)
  if(BF_NATIVE)
    target_compile_options(${target} PRIVATE -march=native)
  endif()
  if(BF_SANITIZERS)
    target_compile_options(${target} PRIVATE -fsanitize=address,undefined -fno-omit-frame-pointer)
    target_link_options(${target} PRIVATE -fsanitize=address,undefined)
  endif()
endforeach()
target_compile_definitions(compare_benchmark PRIVATE BF_EXTERNAL_ONLY=1 BF_EXTERNAL_HEADER="adapters.hpp")
configure_file(manifest.json ${CMAKE_CURRENT_BINARY_DIR}/rocksdb-manifest.json COPYONLY)
include(CTest)
add_test(NAME rocksdb_core_contract COMMAND check)
add_test(NAME rocksdb_harness_contract COMMAND ${Python3_EXECUTABLE}
  ${CMAKE_CURRENT_SOURCE_DIR}/test_contract.py $<TARGET_FILE:compare_benchmark>)
