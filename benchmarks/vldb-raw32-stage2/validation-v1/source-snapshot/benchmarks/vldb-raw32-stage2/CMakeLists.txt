cmake_minimum_required(VERSION 3.16)
project(vldb_raw32_stage2_preflight LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_EXTENSIONS ON)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)
set(VLDB_BACKEND "native" CACHE STRING "native, avx2, native-asan")
get_filename_component(REPO "${CMAKE_CURRENT_SOURCE_DIR}/../.." ABSOLUTE)
set(PARENT "${REPO}/benchmarks/vldb-original-external")
set(DTL "${PARENT}/.deps/dtl")
set(LEGACY_BOOST_ROOT "$ENV{HOME}/.local/boost-1.83/include" CACHE PATH "One complete Boost1.83 header root")
if(NOT CMAKE_CXX_COMPILER_ID STREQUAL "GNU" OR NOT CMAKE_CXX_COMPILER_VERSION VERSION_EQUAL "12.2.0")
  message(FATAL_ERROR "Reused original objects require the validated GCC12.2 compiler; no silent compiler mixing")
endif()
if(VLDB_BACKEND STREQUAL "native")
  set(DEFAULT_ARCHIVE "${REPO}/build/vldb-public-preflight-v2-gcc/native/libvldb_pinned.a")
  set(FLAGS -O3 -march=native)
elseif(VLDB_BACKEND STREQUAL "avx2")
  set(DEFAULT_ARCHIVE "${REPO}/build/vldb-public-preflight-v6-gcc/avx2/libvldb_pinned.a")
  set(FLAGS -O3 -march=x86-64-v3)
elseif(VLDB_BACKEND STREQUAL "native-asan")
  set(DEFAULT_ARCHIVE "${REPO}/build/vldb-public-preflight-v5-gcc/native-asan/libvldb_pinned.a")
  set(FLAGS -O1 -march=native -fsanitize=address,undefined -fno-sanitize-recover=all -fno-omit-frame-pointer)
else()
  message(FATAL_ERROR "Only previously verified whole-library backends are allowed")
endif()
set(VLDB_ARCHIVE "${DEFAULT_ARCHIVE}" CACHE FILEPATH "Exact reused original archive")
set(VLDB_ARCHIVE_SHA256 "" CACHE STRING "Required reviewed archive digest")
if(NOT EXISTS "${VLDB_ARCHIVE}")
  message(FATAL_ERROR "Verified original archive missing; this target never rebuilds it automatically")
endif()
file(SHA256 "${VLDB_ARCHIVE}" ACTUAL_ARCHIVE_SHA256)
if(NOT VLDB_ARCHIVE_SHA256 STREQUAL ACTUAL_ARCHIVE_SHA256)
  message(FATAL_ERROR "Original archive hash missing or changed; use the reviewed preflight driver")
endif()
find_package(Threads REQUIRED)
add_library(vldb_reused STATIC IMPORTED)
set_target_properties(vldb_reused PROPERTIES IMPORTED_LOCATION "${VLDB_ARCHIVE}")
add_executable(vldb_geometry_check geometry_check.cpp)
target_include_directories(vldb_geometry_check PRIVATE "${REPO}/include" "${DTL}/src" "${DTL}/thirdparty" "${LEGACY_BOOST_ROOT}")
target_compile_options(vldb_geometry_check PRIVATE ${FLAGS} -Wno-ignored-attributes -include "${PARENT}/compat.hpp")
target_link_libraries(vldb_geometry_check PRIVATE vldb_reused Threads::Threads ${CMAKE_DL_LIBS})
if(VLDB_BACKEND STREQUAL "native-asan")
  target_link_options(vldb_geometry_check PRIVATE -fsanitize=address,undefined)
endif()
enable_testing()
# Every test is a fresh process. Configure U never mutates an existing tuner key.
foreach(CONFIG sector32-u1 sector32-u2 sector32-u4 sector64-u1 sector64-u2 sector64-u4 register64-u0)
  foreach(POWER 16 25)
    foreach(BPK 12 16 24 32)
      add_test(NAME ${CONFIG}-n${POWER}-b${BPK} COMMAND vldb_geometry_check ${CONFIG} ${POWER} ${BPK})
    endforeach()
  endforeach()
endforeach()
