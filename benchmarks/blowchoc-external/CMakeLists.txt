cmake_minimum_required(VERSION 3.16)
project(blowchoc_bridge LANGUAGES CXX)
option(BLOWCHOC_NATIVE "Compile C++ bridge for this host ISA" ON)
option(BLOWCHOC_SANITIZE "C++ Address/UndefinedBehavior sanitizers" OFF)
add_library(blowchoc_native SHARED native.cpp)
add_executable(check_native check_native.cpp)
target_link_libraries(check_native PRIVATE blowchoc_native)
target_compile_features(check_native PRIVATE cxx_std_17)
target_compile_features(blowchoc_native PRIVATE cxx_std_17)
target_compile_options(blowchoc_native PRIVATE -O3 -Wall -Wextra -Wpedantic)
if(BLOWCHOC_NATIVE)
  target_compile_options(blowchoc_native PRIVATE -march=native)
endif()
if(BLOWCHOC_SANITIZE)
  target_compile_options(blowchoc_native PRIVATE -fsanitize=address,undefined -fno-omit-frame-pointer)
  target_link_options(blowchoc_native PRIVATE -fsanitize=address,undefined)
  target_compile_options(check_native PRIVATE -fsanitize=address,undefined -fno-omit-frame-pointer)
  target_link_options(check_native PRIVATE -fsanitize=address,undefined)
endif()
enable_testing()
add_test(NAME native_bridge COMMAND check_native)
