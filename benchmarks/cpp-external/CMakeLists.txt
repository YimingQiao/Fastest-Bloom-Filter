cmake_minimum_required(VERSION 3.18)
project(fastest_bloom_external LANGUAGES CXX)
if(POLICY CMP0135)
  cmake_policy(SET CMP0135 NEW)
endif()
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)
if(NOT CMAKE_BUILD_TYPE)
  set(CMAKE_BUILD_TYPE Release CACHE STRING "Build type" FORCE)
endif()
option(BF_NATIVE "Give every candidate the build host instruction set" ON)
option(BF_SANITIZERS "Enable address and undefined-behavior checks" OFF)
option(BF_PAGE_AES "Use PageBloom's optional data-incompatible AES-NI hasher" OFF)
include(${CMAKE_CURRENT_SOURCE_DIR}/dependencies.cmake)
add_executable(compare_benchmark ../../src/compare_benchmark.cpp)
target_link_libraries(compare_benchmark PRIVATE FastFilterExternal::options)
target_compile_definitions(compare_benchmark PRIVATE
  BF_EXTERNAL_HEADER="adapters.hpp" BF_EXTERNAL_ONLY=1)
target_compile_options(compare_benchmark PRIVATE -fno-strict-aliasing)
if(BF_NATIVE)
  target_compile_options(compare_benchmark PRIVATE -march=native)
else()
  target_compile_options(compare_benchmark PRIVATE -mavx2)
endif()
if(BF_SANITIZERS)
  target_compile_options(compare_benchmark PRIVATE -fsanitize=address,undefined -fno-omit-frame-pointer)
  target_link_options(compare_benchmark PRIVATE -fsanitize=address,undefined)
endif()
# Persist immutable upstream identities beside the binary, for experiment records.
configure_file(upstreams.json ${CMAKE_CURRENT_BINARY_DIR}/upstreams.json COPYONLY)
include(CTest)
find_package(Python3 COMPONENTS Interpreter REQUIRED)
add_test(NAME external_contract COMMAND ${Python3_EXECUTABLE}
  ${CMAKE_CURRENT_SOURCE_DIR}/test_contract.py $<TARGET_FILE:compare_benchmark>)
